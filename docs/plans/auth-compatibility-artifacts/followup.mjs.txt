const hooks = (tx, mode) => ({
  user: {
    create: {
      after: async (u) => {
        await tx.starter.create({
          data: { userId: u.id, label: "Invented starter" },
        });
        if (mode === "provision-fail")
          throw Error("INVENTED_PROVISION_FAILURE");
      },
    },
  },
  account: {
    create: {
      after: async (a) => {
        if (a.password)
          await tx.user.update({
            where: { id: a.userId },
            data: { password: a.password },
          });
        if (mode === "mirror-fail") throw Error("INVENTED_MIRROR_FAILURE");
      },
    },
    update: {
      after: async (a) => {
        if (mode === "before-mirror-fail")
          throw Error("INVENTED_BEFORE_MIRROR_FAILURE");

});
// The wrapper buffers the response until the outer transaction commits.
// It never exposes a transaction-bound auth instance to a public handler.
const outer = async (kind, body, cookie, mode = "normal") => {
  try {
    const response = await db.$transaction(
      async (tx) => {
        assert.equal(typeof tx.$transaction, "undefined");
        const auth = betterAuth({
          ...options(tx),
          databaseHooks: hooks(tx, mode),
        });
        const response = await auth.api[kind]({
          body,
          headers: new Headers(cookie ? { cookie } : {}),
          asResponse: true,
        });
        if (response.status >= 400)
          throw Object.assign(Error("ENDPOINT_FAILURE"), {
            statusCode: response.status,
          });
        if (mode === "after-endpoint-fail")
          throw Error("INVENTED_AFTER_ENDPOINT_FAILURE");
        return response;
      },
      { timeout: 15000 },
    );
    return { status: response.status, response };
  } catch (error) {
    return {
      status: Number.isInteger(error.statusCode) ? error.statusCode : 500,
      errorName: error.name,
      errorMessage: error.message,
    };
  }
};

// A single guarded SQL statement prevents resurrection and expiry shortening.
const guardedUpdate = async (identity, target) => {
  const changed = await db.$queryRaw(
    Prisma.sql`UPDATE "Session" SET "expiresAt"=GREATEST("expiresAt",${target}),"updatedAt"=GREATEST("updatedAt",clock_timestamp()) WHERE "id"=${identity.id} AND "token"=${identity.token} AND "userId"=${identity.userId} AND "expiresAt">clock_timestamp() RETURNING "expiresAt"`,
  );
  return {
    status: changed.length ? 200 : 401,
    expiresAt: changed[0]?.expiresAt,
  };
};
const renew = async (cookie, target, interleave) => {
  const response = await request(plain, "get-session", undefined, cookie);
  assert.equal(response.status, 200);
  const auth = await response.json();
  if (!auth?.session) return { status: 401 };
  if (interleave) await interleave(auth.session);
  return guardedUpdate(auth.session, target);
};
